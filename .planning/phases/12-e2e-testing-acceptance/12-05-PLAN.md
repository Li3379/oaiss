    /**
     * Verify that an API endpoint exists and is accessible
     */
    export async function verifyEndpointExists(
      method: string,
      path: string,
      token?: string
    ): Promise<{ exists: boolean; status?: number; error?: string }> {
      try {
        const headers: Record<string, string> = {}
        if (token) {
          headers['Authorization'] = `Bearer ${token}`
        }

        let response
        const url = `${API_BASE}${path}`

        switch (method.toUpperCase()) {
          case 'GET':
            response = await playwrightRequest.get(url, { headers })
            break
          case 'POST':
            response = await playwrightRequest.post(url, {
              headers,
              data: {} // Empty body for existence check
            })
            break
          case 'PUT':
            response = await playwrightRequest.put(url, {
              headers,
              data: {}
            })
            break
          case 'DELETE':
            response = await playwrightRequest.delete(url, { headers })
            break
          default:
            return { exists: false, error: `Unknown method: ${method}` }
        }

        // Consider 200, 201, 400, 401, 403, 404 as "endpoint exists"
        // 500+ might indicate endpoint exists but failed
        const status = response.status()
        const exists = status < 500

        return { exists, status }
      } catch (error) {
        return { exists: false, error: String(error) }
      }
    }

    /**
     * Verify coverage for a specific API module
     */
    export async function verifyApiModuleCoverage(
      moduleName: keyof typeof API_MODULE_COVERAGE,
      token?: string
    ): Promise<{ module: string; endpoints: number; covered: number; results: Array<{ endpoint: string; exists: boolean }> }> {
      const endpoints = API_MODULE_COVERAGE[moduleName] || []
      const results: Array<{ endpoint: string; exists: boolean }> = []

      for (const endpoint of endpoints) {
        const [method, path] = endpoint.split(' ')
        const result = await verifyEndpointExists(method, path, token)
        results.push({ endpoint, exists: result.exists })
      }

      const covered = results.filter(r => r.exists).length

      return {
        module: moduleName,
        endpoints: endpoints.length,
        covered,
        results
      }
    }

    /**
     * Get Swagger OpenAPI spec from backend
     */
    export async function getSwaggerSpec(): Promise<Record<string, unknown> | null> {
      try {
        const response = await playwrightRequest.get(`${API_BASE.replace('/api/v1', '')}/v3/api-docs`)
        if (response.ok()) {
          return await response.json()
        }
        return null
      } catch {
        return null
      }
    }

    /**
     * Compare Swagger paths with frontend API modules
     */
    export async function compareSwaggerWithApiModules(): Promise<{
      swaggerPaths: number
      frontendModules: number
      coverage: number
      missingInFrontend: string[]
    }> {
      const swagger = await getSwaggerSpec()
      if (!swagger) {
        return { swaggerPaths: 0, frontendModules: 0, coverage: 0, missingInFrontend: [] }
      }

      const paths = swagger.paths as Record<string, unknown>
      const swaggerPaths = Object.keys(paths).length

      // Count unique frontend endpoints
      let frontendModules = 0
      for (const endpoints of Object.values(API_MODULE_COVERAGE)) {
        frontendModules += endpoints.length
      }

      // Find Swagger paths not in frontend modules (simplified check)
      const missingInFrontend: string[] = []
      for (const path of Object.keys(paths)) {
        // This is a simplified check - full implementation would parse each endpoint
        const normalizedPath = path.replace(/\/api\/v1/, '')
        const hasMatchingModule = Object.values(API_MODULE_COVERAGE).some(endpoints =>
          endpoints.some(e => e.includes(normalizedPath.split('{')[0].split('/')[1]))
        )
        if (!hasMatchingModule && !path.includes('/auth/')) {
          missingInFrontend.push(path)
        }
      }

      const coverage = Math.round((frontendModules / swaggerPaths) * 100)

      return { swaggerPaths, frontendModules, coverage, missingInFrontend }
    }

    /**
     * Read API module file and extract endpoints
     */
    export function extractEndpointsFromApiModule(filePath: string): string[] {
      try {
        const fullPath = path.join(process.cwd(), 'oaiss-chain-frontend', 'src', 'api', filePath)
        const content = fs.readFileSync(fullPath, 'utf-8')

        // Extract request calls (simplified regex)
        const requestRegex = /request\.(get|post|put|delete)\s*\(\s*['"`]([^'"`]+)['"`]/g
        const endpoints: string[] = []
        let match

        while ((match = requestRegex.exec(content)) !== null) {
          endpoints.push(`${match[1].toUpperCase()} ${match[2]}`)
        }

        return endpoints
      } catch {
        return []
      }
    }
