    export class EnterpriseInferencePage {
      constructor(private page: Page) {}

      async goto(enterpriseId?: number): Promise<void> {
        const url = enterpriseId
          ? `/enterprise/inference?enterpriseId=${enterpriseId}`
          : '/enterprise/inference'
        await this.page.goto(url)
        await this.page.waitForLoadState('networkidle')
      }

      async expectLoaded(): Promise<void> {
        await expect(this.page).toHaveURL(/\/enterprise\/inference/)
        await expect(this.page.getByText(/企业推断|Enterprise Inference/)).toBeVisible()
      }

      async expectRiskAssessment(): Promise<void> {
        // Look for risk level display (HIGH/MEDIUM/LOW)
        const riskLabel = this.page.getByText(/风险|Risk Level|合规风险/)
        await expect(riskLabel.first()).toBeVisible()
      }

      async expectEmissionTrendChart(): Promise<void> {
        // Check for ECharts trend chart
        const chart = this.page.locator('div[class*="echarts"]')
        await expect(chart.first()).toBeVisible()
      }

      async expectAnomalyIndicators(): Promise<void> {
        // Look for anomaly detection indicators
        const anomalyLabel = this.page.getByText(/异常|Anomaly/)
        if (await anomalyLabel.isVisible().catch(() => false)) {
          await expect(anomalyLabel.first()).toBeVisible()
        }
      }

      async selectEnterprise(enterpriseName: string): Promise<void> {
        // Try to find enterprise selector
        const selector = this.page.getByRole('combobox').or(
          this.page.getByPlaceholder(/选择企业|Select Enterprise/)
        )
        if (await selector.isVisible().catch(() => false)) {
          await selector.click()
          await this.page.getByText(enterpriseName).click()
        }
      }

      async getRiskLevel(): Promise<string | null> {
        const riskElement = this.page.getByText(/HIGH|MEDIUM|LOW|高|中|低/).first()
        if (await riskElement.isVisible().catch(() => false)) {
          return await riskElement.textContent()
        }
        return null
      }
    }
